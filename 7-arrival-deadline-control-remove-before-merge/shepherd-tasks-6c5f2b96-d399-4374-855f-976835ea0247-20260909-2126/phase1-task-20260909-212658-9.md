                        locationRepository);

        bookingService.assignCargoToRoute(itinerary, trackingId);

        List<CargoRoute> routes = new ArrayList<>(cargos.size());

        CargoRouteDtoAssembler assembler = new CargoRouteDtoAssembler();

        }

        return routeCandidates;
    }
